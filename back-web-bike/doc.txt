    select: {
      user: {
        select: {
          name: true,
          surname: true,
          email: true,
          _count: {
            select: {
              orders: true,
            },
          },
        },
      },

      address: {
        select: {
          country: true,
          city: true,
          street: true,
          phone: true,
        },
      },

      payment: {
        select: {
          idTransaction: true,
          status: true,
        },
      },

      orderItems: {
        select: {
          priceAtPurchase: true,
          quantity: true,

          bike: {
            select: {
              name: true,
              cover: {
                select: {
                  url: true,
                },
              },
            },
          },
        },
      },
    },
